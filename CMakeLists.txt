cmake_minimum_required(VERSION 3.16)
project(ku100_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ku100-native native/main.cpp native/physics.cpp native/receiver.cpp)
if(MSVC)
  target_compile_options(ku100-native PRIVATE /W4)
else()
  target_compile_options(ku100-native PRIVATE -Wall -Wextra -Wpedantic)
endif()
