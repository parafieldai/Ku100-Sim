cmake_minimum_required(VERSION 3.16)
project(Ku100Sim LANGUAGES CXX)
add_executable(ku100_sim src/sim.cpp)
target_compile_features(ku100_sim PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(ku100_sim PRIVATE /W4)
else()
  target_compile_options(ku100_sim PRIVATE -Wall -Wextra -Wpedantic)
endif()
